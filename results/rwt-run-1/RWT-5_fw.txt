def get_middle_third(items: list) -> list:
    """
    Returns the middle third of a list.

    For a list of length N, returns elements from index N//3 to index 2*N//3 (inclusive).

    Args:
        items: The input list.

    Returns:
        A list containing the middle third of the input list.
    """
    n = len(items)
    if n == 0:
        return []

    # Calculate the start and end indices
    # Start index: N // 3
    start_index = n // 3
    
    # End index (exclusive for slicing): 2 * N // 3 + 1
    # Since the requirement is inclusive of 2*N//3, the slice end must be 2*N//3 + 1
    end_index = (2 * n // 3) + 1
    
    # Return the slice
    return items[start_index:end_index]
